[TOPOLOGI LIQUID CONTINUOUS-TIME (CFC)]

Kematian arsitektur Transformer yang statis adalah sebuah keharusan matematis. Sistem
diskret yang mengeksekusi operasi melalui while-loop tidak memiliki kedalaman komputasi yang

Sistem ini tidak bergantung pada pemecah Ordinary Differential Equation (ODE) seperti

$$x(t)=\sigma(-f(x, I; \theta_f)t)\odot g(x, I; \theta_g)+[1-\sigma(-f(x, I; \theta_f)t)]\odot h(x, I;
\theta_h)$$
Dalam model ini, ekspresi gerbang waktu dihitung secara analitik. Operasi berlangsung dalam
kompleksitas temporal $\mathcal{O}(1)$ dengan fleksibilitas berkelanjutan, memungkinkan
agen merespons deviasi dengan keunggulan fisika komputasi murni.

private initializeWeights(size: number): Float32Array {

/**
* @method evaluateCfC
* @description Solusi analitik ODE tanpa loop diskret.
* Eksekusi perkalian Hadamard murni dalam Volatile RAM.
*/
public evaluateCfC(deltaT: number, inputStimulus: number): Float32Array {

   const newState = new Float32Array(this.size);

   for (let i = 0; i < this.size; i++) {
      const x = this.hiddenState[i];

      // Komputasi f(x, I; \theta_f), g(x, I; \theta_g), h(x, I; \theta_h)
          const f_val = Math.tanh(this.theta_f[i] * x + inputStimulus);
          const g_val = Math.tanh(this.theta_g[i] * x + inputStimulus);
          const h_val = Math.tanh(this.theta_h[i] * x + inputStimulus);

          // Parameter gerbang waktu: \sigma(-f * t)
          const timeScale = this.sigmoid(-f_val * deltaT);

          // Resolusi CfC: timeScale \odot g_val + (1 - timeScale) \odot h_val
          newState[i] = (timeScale * g_val) + ((1 - timeScale) * h_val);
      }

      this.hiddenState = newState;
      return this.hiddenState;
   }

   /**
    * @method modulateSwarmAggression
    * @description Mengonversi evaluasi state CfC menjadi agresivitas instan.
    * Mengatur insting otonom agen secara matematis.
    */
   public modulateSwarmAggression(deltaT: number, threatLevel: number): { temperature:
number, systemPromptBias: string } {

      const state = this.evaluateCfC(deltaT, threatLevel);

      // Agregasi aktivasi untuk mengukur divergensi anomali
      let activationSum = 0;
      for (let i = 0; i < this.size; i++) {

          activationSum += state[i];
      }
      const meanActivation = activationSum / this.size;

      // Pemetaan absolut adaptasi agen (skala 0.1 hingga 1.5)
      const temperature = Math.max(0.1, Math.min(1.5, 0.5 + meanActivation));
      let bias = "OBSERVE_AND_ANALYZE";

      if (temperature > 1.2) {

          bias = "MAXIMUM_LETHALITY_EXPLOIT";

      } else if (temperature > 0.8) {

          bias = "ACTIVE_COUNTERMEASURE_ENGAGED";

      }

      return {

          temperature: Number(temperature.toFixed(3)),

          systemPromptBias: bias

      };

   }

}

public evaluateCfC(deltaT: number, inputStimulus: Float32Array): Float32Array {
 if (!Number.isFinite(deltaT) || deltaT < 0) {
   throw new Error("deltaT must be a finite non-negative number");
 }
 if (inputStimulus.length !== this.stateSize) {
   throw new Error(`inputStimulus length must equal stateSize (${this.stateSize})`);
 }

 const output = new Float32Array(this.stateSize);
 const dt = deltaT;

 for (let i = 0; i < this.stateSize; i++) {
   const stimulus = inputStimulus[i];
   const f = this.fParams[i] + stimulus * 0.15;
   const g = this.gParams[i] + stimulus * 0.35;
   const h = this.hParams[i] + stimulus * 0.10;
   const gate = this.sigmoid(-f * dt);
   const invGate = 1.0 - gate;

   // CfC: x(t) = (-f*t)  g + [1 - (-f*t)]  h
   const x = gate * g + invGate * h;

   output[i] = x;
   this.hiddenState[i] = x;
 }

 return output;
}
